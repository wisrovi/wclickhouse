# High-performance analytical ORM
from pydantic import BaseModel
from wclickhouse import WClickHouse

class Event(BaseModel):
    user_id: int
    action: str

db = WClickHouse(Event, db_config)
db.insert_many([Event(user_id=1, action="click")])
